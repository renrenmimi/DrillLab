// 临时路由：把导航所需的轻量元数据 dump 成 JSON，用完即删。
import {
  EXAMS,
  allArena,
  allCodingProblems,
  allDrills,
  lessonsOf,
} from "@/content/registry";
import { PLANS, resolvePlan } from "@/content/plans";

export function GET() {
  return Response.json({
    exams: EXAMS.map((e) => ({
      id: e.id,
      title: e.title,
      titleEn: e.titleEn,
      shortTitle: e.shortTitle,
      shortTitleEn: e.shortTitleEn,
      descriptionEn: e.descriptionEn,
      description: e.description,
      category: e.category,
      tests: e.tests,
      testsEn: e.testsEn,
      stack: e.stack,
      status: e.status,
      prerequisites: e.prerequisites,
      sourceProjects: e.sourceProjects,
      lessonCount: lessonsOf(e).length,
      exerciseCount: lessonsOf(e).reduce((n, r) => n + (r.lesson.exercises?.length ?? 0), 0),
      minutes: lessonsOf(e).reduce((n, r) => n + r.lesson.minutes, 0),
      debugLabs: lessonsOf(e).reduce(
        (n, r) => n + (r.lesson.exercises ?? []).filter((x) => x.kind === "debug").length, 0),
      rebuilds: lessonsOf(e).reduce(
        (n, r) => n + (r.lesson.exercises ?? []).filter((x) => x.kind === "from-scratch").length, 0),
      rebuildIds: lessonsOf(e).flatMap((r) =>
        (r.lesson.exercises ?? []).filter((x) => x.kind === "from-scratch").map((x) => x.id)),
      // 【为什么这两张小表值得进 nav】Practice 模式的侧栏要在「按题型 / 按难度」
      // 那两排筛选项上写出数量，而它是客户端组件，数不了 allExercises()。
      // 加起来是每门课 10 个整数（6 种题型 + 4 个难度），可以忽略不计。
      exerciseKinds: lessonsOf(e).reduce((acc, r) => {
        for (const x of r.lesson.exercises ?? []) acc[x.kind] = (acc[x.kind] ?? 0) + 1;
        return acc;
      }, {}),
      exerciseLevels: lessonsOf(e).reduce((acc, r) => {
        for (const x of r.lesson.exercises ?? []) {
          const k = String(x.level);
          acc[k] = (acc[k] ?? 0) + 1;
        }
        return acc;
      }, {}),
      modules: e.modules.map((m) => ({
        id: m.id,
        title: m.title,
        titleEn: m.titleEn,
        summary: m.summary,
        summaryEn: m.summaryEn,
        stage: m.stage,
        // 【只放客户端首屏真的要用的】其余全部搬去下面的 searchIndex —— 见 nav-template 的注释
        lessons: m.lessons.map((l) => ({
          id: l.id,
          title: l.title,
          titleEn: l.titleEn,
          blurb: l.blurb,
          blurbEn: l.blurbEn,
          minutes: l.minutes,
          exerciseCount: l.exercises?.length ?? 0,
        })),
      })),
      mockExams: e.mockExams.map((m) => ({
        id: m.id,
        title: m.title,
        titleEn: m.titleEn,
        scenario: m.scenario,
        mirrors: m.mirrors,
        minutes: m.minutes,
        taskCount: m.tasks.length,
        taskTitles: m.tasks.map((t) => t.title),
        outOf: m.tasks.reduce((n, t) => n + t.rubric.reduce((k, r) => k + r.points, 0), 0),
      })),
      checklist: e.checklist ?? [],
    })),

    // 八股题清单 —— 客户端的筛选、抽认卡、搜索只读这个，
    // 绝对不许把 content/drills 拉进客户端组件（单 chunk 784 KB 的坑）。
    drills: allDrills().map((q) => ({
      id: q.id,
      bank: q.bank,
      zh: q.zh,
      en: q.en,
      track: q.track,
      lessonId: q.lessonId,
      examId: q.examId,
      hasEn: !!q.answerEn,
    })),

    coding: allCodingProblems().map((p) => ({
      id: p.id,
      title: p.title,
      titleEn: p.titleEn,
      track: p.track,
      difficulty: p.difficulty,
      minutes: p.minutes,
      runnable: p.runnable,
      hasSandbox: !!p.sandbox,
      explainLessonId: p.explainLessonId,
      requirementCount: p.requirements.length,
    })),

    arena: allArena().map((a) => ({
      id: a.id,
      title: a.title,
      titleEn: a.titleEn,
      scenarioEn: a.scenarioEn,
      scenario: a.scenario,
      minutes: a.minutes,
      examId: a.sourceExamId,
      fromMock: !!a.sourceMockId,
      requirementCount: a.requirements.length,
      commandCount: a.commands.length,
      runnable: !!a.blankSandbox,
    })),

    /**
     * 首页仪表盘用的轨道清单 —— 生成到 content/track-manifest.ts。
     * 只有课程名、一句话、和按顺序排好的课文（id / 标题 / 链接）。
     * 理由见那个文件顶部。
     */
    trackManifest: {
      tracks: EXAMS.map((e) => ({
        id: e.id,
        zh: e.shortTitle,
        en: e.shortTitleEn,
        blurbZh: e.description,
        blurbEn: e.descriptionEn,
        // 平行支线的判据和 content/path.ts 完全一致：自己没有前置，
        // 而且没有别的课把它列成前置。抄第二套判据会让首页和 /path 分歧。
        parallel:
          e.prerequisites.length === 0 &&
          !EXAMS.some((o) => o.prerequisites.includes(e.id))
            ? true
            : undefined,
        lessons: lessonsOf(e).map((r) => ({
          id: r.lesson.id,
          href: `/exams/${e.id}/${r.lesson.id}`,
          zh: r.lesson.title,
          en: r.lesson.titleEn,
        })),
      })),
      surfaces: {
        drills: allDrills().length,
        exercises: EXAMS.reduce(
          (n, e) => n + lessonsOf(e).reduce((k, r) => k + (r.lesson.exercises?.length ?? 0), 0), 0),
        coding: allCodingProblems().length,
        arena: allArena().length,
        mocks: EXAMS.reduce((n, e) => n + e.mockExams.length, 0),
      },
    },

    /**
     * 引导计划的轻量清单 —— 生成到 content/plan-manifest.ts。
     *
     * 【为什么要在构建期把它压出来】
     * 展开一条计划要读 content/nav（134 KB）+ content/nav-exercises（39 KB）。
     * 而顶栏徽标、侧栏面板、页内位置条是全站挂载的，它们顺着 plan-kit 拖上
     * content/plans 的后果是 webpack 把 nav 拆成独立 chunk、每个路由各下一遍
     * —— 实测课程页 First Load JS 142 → 195 kB。
     *
     * 所以在这里展开一次，只留「算我在哪、下一格是什么」要的字段：
     * 条目去重共享一张表，每条计划只存下标。
     * 「为什么在这儿」、每条的估时、覆盖方向都不进去 —— 那些只有
     * /plans/[planId] 用，那一页自己读 content/plans。
     */
    planManifest: (() => {
      const index = new Map();
      const items = [];
      const plans = PLANS.map((p) => {
        const r = resolvePlan(p);
        return {
          id: p.id,
          zh: p.zh,
          en: p.en,
          outcomeZh: p.outcomeZh,
          outcomeEn: p.outcomeEn,
          forZh: p.forZh,
          forEn: p.forEn,
          minutes: r.minutes,
          stages: r.stages.map((st) => ({
            id: st.id,
            phase: st.phase,
            zh: st.zh,
            en: st.en,
            // 「这一档为什么在这儿」是计划**唯一自己拥有的文字**。
            // 首页那张「下一件事」的卡要用它回答「为什么是这一件」，
            // 所以它必须进轻量清单（45 档 × 两种语言，约 10 KB）。
            whyZh: st.whyZh,
            whyEn: st.whyEn,
            items: st.items.map((it) => {
              if (!index.has(it.key)) {
                index.set(it.key, items.length);
                items.push([
                  it.kind,
                  it.id,
                  it.examId ?? "",
                  it.zh,
                  it.en ?? "",
                  it.href,
                  it.exerciseKind ?? "",
                  it.minutes ?? 0,
                ]);
              }
              return index.get(it.key);
            }),
          })),
        };
      });
      return { items, plans };
    })(),

    /**
     * 练习清单 —— 单独一份，生成到 content/nav-exercises.ts。
     *
     * 【为什么不放进 nav.ts】
     * 引导计划里「做练习」那一档的每一格都是一个真实练习，完成状态读
     * progress.exercises[`${examId}/${id}`]，所以计划要拿到这 148 个 id。
     * 但 nav.ts 被**每一个客户端页面**下载，而计划那一套是懒加载的
     * （只有真的在跟计划的人才下载，见 components/plan-slots.tsx）。
     *
     * 把练习塞进 nav.ts 实测的代价：nav 从 133 KB 长到 160 KB，
     * webpack 的分块随之改变，课程页从 470 kB 原始字节涨到 615 kB ——
     * 而课程页原本一个字节的 nav 都不下载。
     * 分成两份之后 nav.ts 和 main 逐字节一致，这一份只进那个异步 chunk。
     */
    exercises: EXAMS.flatMap((e) =>
      lessonsOf(e).flatMap((r) =>
        (r.lesson.exercises ?? []).map((x) => ({
          examId: e.id,
          lessonId: r.lesson.id,
          id: x.id,
          title: x.title,
          titleEn: x.titleEn,
          kind: x.kind,
          level: x.level,
        })),
      ),
    ),

    /**
     * 搜索专用的重字段 —— 单独一份，生成到 content/search-index.ts。
     *
     * 这些字段加起来 130 KB 出头，而它们只有一个消费者：搜索的 haystack。
     * 放在 nav.ts 里会让每个页面都白下 80 kB（gzip 后）。
     * search.tsx 在用户第一次打开搜索时才 import() 它。
     */
    searchIndex: EXAMS.flatMap((e) =>
      e.modules.flatMap((m) =>
        m.lessons.map((l) => ({
          examId: e.id,
          lessonId: l.id,
          objectives: l.objectives,
          whyForAssessment: l.whyForAssessment,
          conceptHeadings: l.concepts.map((c) => c.heading),
          // lede 里放的是英文原题和题库编号（面试八股那门课），搜索要能命中它
          conceptLedes: l.concepts.map((c) => c.lede ?? "").filter(Boolean),
          exerciseTitles: (l.exercises ?? []).map((x) => x.title),
          sourcePaths: (l.sourceFiles ?? []).map((f) => f.path),
          recap: l.recap ?? [],
          transfer: (l.transfer ?? []).map((t) => `${t.signal} ${t.reachFor}`),
        })),
      ),
    ),
  });
}
